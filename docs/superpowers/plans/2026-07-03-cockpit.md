# 總管駕駛艙（Cockpit）實現計劃

> **面向 AI 代理的工作者：** 必需子技能：使用 superpowers:subagent-driven-development（推薦）或 superpowers:executing-plans 逐任務實現此計劃。步驟使用複選框（`- [ ]`）語法追蹤進度。

**目標：** 每天固定時間生成一份跨所有工作區的「總管簡報」（待拍板／新項目／各事業近況／學習治理），在 dashboard 內主動迎面而來，每項可展開討論、學習提案可一鍵處理。

**架構：** server 端 `cockpitContext` 受控組出跨工作區資料包 → 隔離的幕僚長 agent 讀包產出結構化簡報 JSON → 解析存 `cockpit_briefings` → 前端 `CockpitPanel` 渲染四區塊。跨工作區資料由 server 主動組包（非 agent 爬沙箱），幕僚長跑在 repo 外隔離 sandbox，不破壞既有工作區記憶隔離。

**技術棧：** Node + TypeScript + node:sqlite（`node:sqlite`）、Express、node-cron、React/Vite、Vitest。規格見 `docs/superpowers/specs/2026-07-03-cockpit-design.md`。

```ts
import { db } from "./db.js";
import { v4 as uuid } from "uuid";
import type { Briefing, BriefingRow } from "./cockpitTypes.js";

export function saveBriefing(input: {
  date: string; content: Briefing | null; raw: string;
  status: BriefingRow["status"];
}): BriefingRow {
  const row: BriefingRow = {
    id: uuid(), date: input.date, content: input.content,
    raw: input.raw, status: input.status, createdAt: Date.now(),
  };
  db.prepare(
    `INSERT INTO cockpit_briefings (id, date, content, raw, status, created_at)
     VALUES (?, ?, ?, ?, ?, ?)`
  ).run(row.id, row.date, row.content ? JSON.stringify(row.content) : null,
        row.raw, row.status, row.createdAt);
  return row;
}

export function getLatestBriefing(): BriefingRow | null {
  const r = db.prepare(
    `SELECT * FROM cockpit_briefings ORDER BY created_at DESC LIMIT 1`
  ).get() as any;
  if (!r) return null;
  return {
    id: r.id, date: r.date,
    content: r.content ? JSON.parse(r.content) as Briefing : null,
    raw: r.raw, status: r.status, createdAt: r.created_at,
  };
}
```

`server/src/cockpitContext.ts`：
```ts
import { listWorkspaces, listSessions, DEFAULT_WORKSPACE_ID } from "./store.js";
import { listPendingProposals } from "./learningStore.js";
import type { CockpitContextPack } from "./cockpitTypes.js";

const MEMO_EXCERPT = 800;

/** 取該工作區最近一份「[排程]」session 的標題（重點來源，MVP 先用標題+時間）。 */
function latestReportFor(workspaceId: string): string | undefined {
  const sessions = listSessions(workspaceId)
    .filter((s) => (s.title || "").includes("排程"));
  if (!sessions.length) return undefined;
  const s = sessions[0]; // listSessions 依 created_at DESC
  return s.title;
}

/** 組跨工作區資料包。純資料彙整、無 LLM。只納入「有記憶內容」的工作區。 */
export function buildCockpitContext(): CockpitContextPack {
  const workspaces = listWorkspaces()
    .filter((w) => !w.id.startsWith("__"))            // 排除隱藏 cockpit 工作區
    .map((w) => ({
      id: w.id,
      name: w.name,
      memoryExcerpt: (w.memory || "").slice(0, MEMO_EXCERPT),
      latestReport: latestReportFor(w.id),
    }))
    .filter((w) => w.memoryExcerpt.trim() || w.latestReport); // 略過完全沒料的

  const pendingProposals = listPendingProposals(DEFAULT_WORKSPACE_ID)
    .map((p) => ({ id: p.id, agentId: p.agentId, kind: p.kind, content: p.content }));

  return {
    generatedAtIso: new Date().toISOString(),  // 註：測試環境 new Date() 可用；此檔非 workflow 腳本
    workspaces,
    pendingProposals,
    staleTopics: [],
  };
}
```
> 註：`listPendingProposals(DEFAULT_WORKSPACE_ID)` 會回「該工作區 + agent-global + category」的 pending（見 learningStore 實作）；能力學習提案多為 agent-global，故涵蓋。若日後要跨全部工作區的 workspace-scope 事實提案，另迭代。

```ts
import type { Briefing, BriefingSection } from "./cockpitTypes.js";

const VALID_KINDS = new Set(["decisions", "new_items", "status", "learning"]);

export function parseBriefing(
  raw: string, date: string, generatedAt: number,
): { status: "ok" | "parse_failed"; briefing: Briefing | null } {
  const m = raw.match(/```json\s*([\s\S]*?)```/i);
  const jsonText = m ? m[1] : null;
  if (!jsonText) return { status: "parse_failed", briefing: null };
  let obj: any;
  try { obj = JSON.parse(jsonText.trim()); }
  catch { return { status: "parse_failed", briefing: null }; }
  if (!obj || !Array.isArray(obj.sections)) {
    return { status: "parse_failed", briefing: null };
  }
  const sections: BriefingSection[] = obj.sections
    .filter((s: any) => s && VALID_KINDS.has(s.kind) && Array.isArray(s.items))
    .map((s: any) => ({
      kind: s.kind,
      title: String(s.title || ""),
      items: s.items.map((it: any) => ({
        title: String(it.title || ""),
        body: String(it.body || ""),
        workspaceId: it.workspaceId ? String(it.workspaceId) : undefined,
        discussSeed: it.discussSeed ? String(it.discussSeed) : undefined,
        action: it.action && (it.action.type === "approve" || it.action.type === "reject")
          && Array.isArray(it.action.proposalIds)
          ? { type: it.action.type, proposalIds: it.action.proposalIds.map(String) }
          : undefined,
      })),
    }));
  return { status: "ok", briefing: { date, generatedAt, sections } };
}
```

```ts
import { agentManager } from "./agentManager.js";
import { getWorkspace, createWorkspace } from "./store.js";
import { buildCockpitContext } from "./cockpitContext.js";
import { parseBriefing } from "./cockpitParser.js";
import { saveBriefing } from "./cockpitStore.js";
import type { CockpitContextPack, BriefingRow } from "./cockpitTypes.js";

export const COCKPIT_WORKSPACE_ID = "__cockpit__";
export const COCKPIT_AGENT_ID = "chief-of-staff"; // 步驟 1 若確認不同，改這裡

/** 確保隱藏 cockpit 工作區存在（repo 外隔離 sandbox；不供 UI 顯示）。 */
function ensureCockpitWorkspace(): void {
  if (!getWorkspace(COCKPIT_WORKSPACE_ID)) {
    createWorkspace({ name: "（總管駕駛艙）", id: COCKPIT_WORKSPACE_ID } as any);
  }
}

function buildPrompt(pack: CockpitContextPack): string {
  return `你是使用者的幕僚長。以下是今天各事業與學習系統的現況資料包（由系統彙整，不需你再去查）。請綜觀全局，產出一份「今日總管簡報」。

# 資料包
${JSON.stringify(pack, null, 2)}

# 輸出要求
在回覆最後輸出一段 \`\`\`json，符合以下結構（sections 每項 kind 從 decisions/new_items/status/learning 四選一）：
{
  "sections": [
    { "kind": "decisions", "title": "🔴 待你拍板", "items": [ { "title": "...", "body": "你的說明與建議", "workspaceId": "對應工作區id", "discussSeed": "展開討論時要接續的一句脈絡" } ] },
    { "kind": "new_items", "title": "🆕 我建議討論的新項目", "items": [ ... ] },
    { "kind": "status", "title": "📋 各事業近況 + 本週重點", "items": [ ... ] },
    { "kind": "learning", "title": "🎓 學習 & 記憶治理", "items": [ { "title": "本週提案 triage", "body": "建議批准/駁回哪些+理由", "action": { "type": "approve", "proposalIds": ["..."] } } ] }
  ]
}
規則：learning 區塊要就資料包裡的 pendingProposals 給 triage 建議，把建議批准的 id 放進 type=approve 的 action、建議駁回的放 type=reject（可拆成兩個 item）。貼合使用者實際業務用途的內容手藝建議批准；與其用途不符（如純軟體開發流水線手藝）建議駁回。沒有待處理項目的區塊就不要放。`;
}

/** 預設 runAgent：在隔離 cockpit 工作區開幕僚長 session，收最終輸出。 */
async function defaultRunAgent(prompt: string): Promise<string> {
  ensureCockpitWorkspace();
  return new Promise<string>((resolve, reject) => {
    const session = agentManager.start(
      COCKPIT_AGENT_ID, "🧭 產生總管簡報", undefined, COCKPIT_WORKSPACE_ID, false,
    );
    let collected = "", streamed = "", settled = false;
    const done = (v: string) => { if (settled) return; settled = true; clearTimeout(t); session.removeListener("event", onEvent); resolve(v); };
    const onEvent = (evt: any) => {
      if (evt.type === "delta" && typeof evt.payload === "string") streamed += evt.payload;
      else if (evt.type === "message" && evt.payload?.content) collected = String(evt.payload.content);
      else if (evt.type === "result") done((collected || streamed).trim());
      else if (evt.type === "error") { /* 收集但等 result/timeout */ }
    };
    const t = setTimeout(() => done((collected || streamed).trim()), 5 * 60 * 1000);
    session.on("event", onEvent);
    agentManager.send(session.id, prompt);
  }).catch((e) => { throw e; });
}

export interface CockpitDeps { runAgent: (prompt: string) => Promise<string>; }

/** 主流程：組包→跑幕僚長→解析→存。永不 throw，失敗存 status=error。 */
export async function runCockpit(
  trigger: "daily" | "manual",
  deps: CockpitDeps = { runAgent: defaultRunAgent },
): Promise<BriefingRow> {
  const date = new Date().toISOString().slice(0, 10);
  const generatedAt = Date.now();
  try {
    const pack = buildCockpitContext();
    const raw = await deps.runAgent(buildPrompt(pack));
    const { status, briefing } = parseBriefing(raw, date, generatedAt);
    return saveBriefing({ date, content: briefing, raw, status });
  } catch (e: any) {
    return saveBriefing({ date, content: null, raw: String(e?.message || e), status: "error" });
  }
}
```
> 註：`createWorkspace` 需支援帶入固定 `id`。若現行簽名不吃 `id`，任務 4 增加一小步：讓 `createWorkspace` 接受可選 `id`（`store/workspaces.ts`，`input.id || uuid()`），並補一條 store 測試。

```tsx
import { useEffect, useState } from "react";
import { api, type BriefingRow, type BriefingItem } from "../lib/api";
import { setActiveWorkspace } from "../lib/workspace";

interface Props {
  onDiscuss: (item: BriefingItem) => void;  // 展開討論（切工作區+開 session）由 App 處理
  onReloadWorkspaces?: () => void;
}

export function CockpitPanel({ onDiscuss }: Props) {
  const [row, setRow] = useState<BriefingRow | null>(null);
  const [busy, setBusy] = useState(false);

  const load = () => api.cockpit().then((r) => setRow(r.briefing)).catch(() => {});
  useEffect(() => { load(); }, []);

  const refresh = async () => {
    setBusy(true);
    try { const r = await api.cockpitRefresh(); setRow(r.briefing); }
    finally { setBusy(false); }
  };

  const applyAction = async (item: BriefingItem) => {
    if (!item.action) return;
    const path = item.action.type === "approve" ? "bulk-approve" : "bulk-reject";
    await fetch(`/api/learning/proposals/${path}`, {
      method: "POST", headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ ids: item.action.proposalIds }),
    });
    load();
  };

  if (!row) return <div className="p-6 text-zinc-500">尚無今日簡報。<button onClick={refresh} className="underline ml-2">立即生成</button></div>;

  return (
    <div className="p-4 md:p-6 max-w-3xl mx-auto space-y-5 overflow-y-auto">
      <div className="flex items-center justify-between">
        <h1 className="text-lg font-semibold">🧭 今日總管簡報</h1>
        <div className="text-xs text-zinc-500 flex items-center gap-2">
          <span>更新：{new Date(row.createdAt).toLocaleString("zh-TW", { hour12: false })}</span>
          <button onClick={refresh} disabled={busy} className="px-2 py-1 rounded bg-zinc-800 hover:bg-zinc-700 disabled:opacity-40">🔄 再想一次</button>
        </div>
      </div>

      {row.status !== "ok" && (
        <div className="text-xs text-amber-400 border border-amber-800/40 rounded p-2">
          簡報格式待修（status={row.status}），以下為原始輸出：
          <pre className="whitespace-pre-wrap mt-1 text-zinc-300">{row.raw.slice(0, 4000)}</pre>
        </div>
      )}

      {row.content?.sections.map((sec, i) => (
        <section key={i} className="space-y-2">
          <h2 className="text-sm font-medium text-zinc-300">{sec.title}</h2>
          {sec.items.map((it, j) => (
            <div key={j} className="rounded-lg border border-zinc-800 bg-panel p-3">
              <div className="font-medium text-sm">{it.title}</div>
              <div className="text-sm text-zinc-400 mt-1 whitespace-pre-wrap">{it.body}</div>
              <div className="flex gap-2 mt-2">
                <button onClick={() => onDiscuss(it)} className="text-xs px-2 py-1 rounded bg-accent hover:bg-violet-500 text-white">💬 展開討論</button>
                {it.action && (
                  <button onClick={() => applyAction(it)} className="text-xs px-2 py-1 rounded bg-emerald-700 hover:bg-emerald-600 text-white">
                    照建議{it.action.type === "approve" ? "批准" : "駁回"}（{it.action.proposalIds.length}）
                  </button>
                )}
              </div>
            </div>
          ))}
        </section>
      ))}
    </div>
  );
}
```
